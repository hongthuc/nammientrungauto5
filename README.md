        function handleSearch(keyword) {
            const q = (keyword || '').trim().toLowerCase();
            if (!q) {
                renderProducts(PRODUCTS_DATA);
                return;
            }
            const filtered = PRODUCTS_DATA.filter(p =>
                p.name.toLowerCase().includes(q) ||
                p.sku.toLowerCase().includes(q) ||
                p.desc.toLowerCase().includes(q)
            );
            renderProducts(filtered);
        }

        /* Điều khiển 2 Banner Tự Trượt (Auto Slider 5 giây - Giữ nguyên chiều cao bằng nhau) */
        let currentSlideIndex = 0;
        let sliderTimer = null;

        function goToBannerSlide(index) {
            currentSlideIndex = index;
            const slide0 = document.getElementById('banner-slide-0');
            const slide1 = document.getElementById('banner-slide-1');
            const dot0 = document.getElementById('slider-dot-0');
            const dot1 = document.getElementById('slider-dot-1');

            if (index === 0) {
                slide0.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
                slide0.classList.add('opacity-100', 'z-10', 'pointer-events-auto');
                slide1.classList.remove('opacity-100', 'z-10', 'pointer-events-auto');
                slide1.classList.add('opacity-0', 'z-0', 'pointer-events-none');
                dot0.className = 'w-6 h-2 rounded-full bg-white transition-all';
                dot1.className = 'w-2 h-2 rounded-full bg-white/50 transition-all';
            } else {
                slide0.classList.remove('opacity-100', 'z-10', 'pointer-events-auto');
                slide0.classList.add('opacity-0', 'z-0', 'pointer-events-none');
                slide1.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
                slide1.classList.add('opacity-100', 'z-10', 'pointer-events-auto');
                dot0.className = 'w-2 h-2 rounded-full bg-white/50 transition-all';
                dot1.className = 'w-6 h-2 rounded-full bg-white transition-all';
            }
        }

        function nextBannerSlide() {
            goToBannerSlide(currentSlideIndex === 0 ? 1 : 0);
        }

        function prevBannerSlide() {
            goToBannerSlide(currentSlideIndex === 0 ? 1 : 0);
        }
